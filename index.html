/** * CAMPAIGN HUB — GOOGLE SHEETS BACKEND * ------------------------------------- * This turns a Google Sheet into a simple free database for the Campaign Hub * dashboard, so the dashboard can be hosted anywhere (GitHub Pages, Netlify, * your own domain) instead of relying on Claude's built-in storage. * * SETUP (one-time, ~5 minutes): * 1. Go to sheets.google.com and create a new blank Sheet. Name it anything, * e.g. "Campaign Hub Data". Leave it empty — this script manages the rows. * 2. In that Sheet, go to Extensions > Apps Script. * 3. Delete any starter code in the editor and paste this entire file in. * 4. Click Deploy > New deployment. * - Click the gear icon next to "Select type" and choose "Web app". * - Description: anything, e.g. "Campaign Hub API". * - Execute as: Me. * - Who has access: Anyone. * - Click Deploy. Authorize when prompted (it's your own script, on your * own account — this is expected). * 5. Copy the "Web app URL" it gives you (ends in /exec). * 6. Paste that URL into campaign-hub.html, replacing the placeholder: * const API_URL = "PASTE_YOUR_APPS_SCRIPT_WEB_APP_URL_HERE"; * * If you ever edit this script again, you must create a NEW deployment * version (Deploy > Manage deployments > Edit > New version) for changes * to take effect on the same URL. * * This version also handles screenshot uploads (people who don't know how * to get a shareable link can just upload the image directly). Uploaded * files are saved into a Google Drive folder called "Campaign Hub * Screenshots" that this script creates automatically the first time * someone uploads. When you redeploy after adding this, Google will ask * you to re-authorize with an extra Drive permission — that's expected. */ const SHEET_NAME = "Data"; function getSheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(["key", "value", "updatedAt"]); } return sheet; } function findRow_(sheet, key) { const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] === key) return i + 1; // 1-indexed row number } return -1; } function doGet(e) { const action = e.parameter.action; const sheet = getSheet_(); try { if (action === "get") { const row = findRow_(sheet, e.parameter.key); if (row === -1) return jsonOut_({ value: null }); const value = sheet.getRange(row, 2).getValue(); return jsonOut_({ value: value }); } if (action === "set") { const key = e.parameter.key; const value = e.parameter.value; const row = findRow_(sheet, key); if (row === -1) { sheet.appendRow([key, value, new Date().toISOString()]); } else { sheet.getRange(row, 2).setValue(value); sheet.getRange(row, 3).setValue(new Date().toISOString()); } return jsonOut_({ ok: true }); } if (action === "delete") { const row = findRow_(sheet, e.parameter.key); if (row !== -1) sheet.deleteRow(row); return jsonOut_({ ok: true }); } if (action === "list") { const prefix = e.parameter.prefix || ""; const data = sheet.getDataRange().getValues(); const keys = []; for (let i = 1; i < data.length; i++) { if (String(data[i][0]).indexOf(prefix) === 0) keys.push(data[i][0]); } return jsonOut_({ keys: keys }); } return jsonOut_({ error: "Unknown action" }); } catch (err) { return jsonOut_({ error: String(err) }); } } function jsonOut_(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } const UPLOAD_FOLDER_NAME = "Campaign Hub Screenshots"; function getUploadFolder_() { const folders = DriveApp.getFoldersByName(UPLOAD_FOLDER_NAME); if (folders.hasNext()) return folders.next(); return DriveApp.createFolder(UPLOAD_FOLDER_NAME); } /** * Handles screenshot uploads from the entry form. * Expects a POST body (Content-Type: text/plain, to dodge CORS preflight) * containing JSON: { action: "uploadImage", filename, mimeType, base64 } * base64 should NOT include the "data:image/png;base64," prefix — the * front-end strips that before sending. */ function doPost(e) { try { const body = JSON.parse(e.postData.contents); if (body.action !== "uploadImage") { return jsonOut_({ error: "Unknown POST action" }); } const folder = getUploadFolder_(); const bytes = Utilities.base64Decode(body.base64); const blob = Utilities.newBlob(bytes, body.mimeType || "image/png", body.filename || ("screenshot-" + Date.now())); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); // file.getUrl() returns a Drive VIEWER page, not a raw image — that breaks // when used as an . This format serves the actual image bytes. const directUrl = "https://drive.google.com/thumbnail?id=" + file.getId() + "&sz=w2000"; return jsonOut_({ url: directUrl, fileId: file.getId() }); } catch (err) { return jsonOut_({ error: String(err) }); } }
